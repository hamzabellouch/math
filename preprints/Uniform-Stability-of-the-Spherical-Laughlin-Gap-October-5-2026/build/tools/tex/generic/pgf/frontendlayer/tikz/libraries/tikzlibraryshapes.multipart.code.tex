    \pgfkeys{/pgf/rectangle split parts={#1}}%
    }%

    \pgfkeys{/pgf/rectangle split use custom fill={#1}}%

}%

    }%
}%

        \pgfkeys{/pgf/rectangle split part fill={#1}}%
    }%
}%
